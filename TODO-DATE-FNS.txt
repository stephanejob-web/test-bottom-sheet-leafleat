Fichier : utils/dateHelpers.ts (à créer)
----------------------------------------

import { formatDistanceToNow, format, parseISO } from 'date-fns';
import { fr } from 'date-fns/locale';

/**
 * Retourne le temps relatif avant/après un événement
 * @param dateString - Date au format ISO string
 * @returns "dans 12 jours" ou "il y a 3 jours"
 */
export const getRelativeEventTime = (dateString: string): string => {
  const eventDate = parseISO(dateString);

  return formatDistanceToNow(eventDate, {
    addSuffix: true,  // Ajoute "dans" ou "il y a"
    locale: fr,       // Texte en français
  });
};

/**
 * Formatte une date complète en français
 * @param dateString - Date au format ISO string
 * @returns "vendredi 20 décembre 2025"
 */
export const formatEventDate = (dateString: string): string => {
  const date = parseISO(dateString);

  return format(date, "EEEE d MMMM yyyy", { locale: fr });
};

/**
 * Formatte une date avec heure
 * @param dateString - Date au format ISO string
 * @param timeString - Heure au format "HH:mm"
 * @returns "ven. 20 déc. à 14h30"
 */
export const formatEventDateTime = (dateString: string, timeString: string): string => {
  const date = parseISO(dateString);

  return format(date, "EEE d MMM", { locale: fr }) + ` à ${timeString}`;
};

  ÉTAPE 7 : AMÉLIORATION OPTIONNELLE - BADGE DE COULEUR

Fonction pour déterminer la couleur selon la proximité :
---------------------------------------------------------

export const getEventUrgencyColor = (dateString: string): string => {
  const eventDate = parseISO(dateString);
  const now = new Date();
  const daysUntil = Math.floor((eventDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24));

  if (daysUntil < 0) return '#94A3B8'; // Passé - Gris
  if (daysUntil <= 2) return '#EF4444'; // Urgent - Rouge
  if (daysUntil <= 7) return '#F59E0B'; // Bientôt - Orange
  return '#10B981'; // Normal - Vert
};

Utilisation :
-------------
<View style={[styles.relativeTimeBadge, { backgroundColor: getEventUrgencyColor(selectedItem.date) + '20' }]}>
  <MaterialCommunityIcons
    name="clock-outline"
    size={14}
    color={getEventUrgencyColor(selectedItem.date)}
  />
  <Text style={[styles.relativeTimeText, { color: getEventUrgencyColor(selectedItem.date) }]}>
    {getRelativeEventTime(selectedItem.date)}
  </Text>
</View>
